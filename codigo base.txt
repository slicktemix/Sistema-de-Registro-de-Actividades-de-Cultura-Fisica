<script>

function toggleEvent(element){
    element.classList.toggle('active');
}

const womenInputs = document.querySelectorAll('.women');
const menInputs = document.querySelectorAll('.men');

function calculateTotals(){

    const rows = document.querySelectorAll('tbody tr');

    rows.forEach(row=>{

        const women = parseInt(row.querySelector('.women').value) || 0;
        const men = parseInt(row.querySelector('.men').value) || 0;

        row.querySelector('.total').value = women + men;

    });

}

womenInputs.forEach(input=>{
    input.addEventListener('input',calculateTotals);
});

menInputs.forEach(input=>{
    input.addEventListener('input',calculateTotals);
});

function generatePDF(){

    const selectedEvents = [];

    document.querySelectorAll('.event-card.active').forEach(card=>{
        selectedEvents.push(card.innerText);
    });

    const rows = document.querySelectorAll('tbody tr');

    let participantRows = '';

    rows.forEach(row=>{

        const categoria = row.cells[0].innerText;
        const women = row.querySelector('.women').value || 0;
        const men = row.querySelector('.men').value || 0;
        const total = row.querySelector('.total').value || 0;

        participantRows += `
            <tr>
                <td>${categoria}</td>
                <td>${women}</td>
                <td>${men}</td>
                <td>${total}</td>
            </tr>
        `;

    });

    const pdfContent = `
    
    <div class="pdf-template">

        <div class="pdf-header">

            <h4>Universidad Autónoma del Estado de México</h4>
            <h3>REGISTRO DE ACTIVIDADES DE CULTURA FÍSICA</h3>

        </div>

        <table class="pdf-table">

            <tr>
                <th>Fecha</th>
                <td>${document.getElementById('fecha').value}</td>
            </tr>

            <tr>
                <th>Espacio Académico</th>
                <td>${document.getElementById('espacio').value}</td>
            </tr>

            <tr>
                <th>Nombre del evento</th>
                <td>${document.getElementById('evento').value}</td>
            </tr>

            <tr>
                <th>Organizador</th>
                <td>${document.getElementById('organizador').value}</td>
            </tr>

            <tr>
                <th>Lugar del evento</th>
                <td>${document.getElementById('lugar').value}</td>
            </tr>

            <tr>
                <th>Duración</th>
                <td>${document.getElementById('duracion').value}</td>
            </tr>

            <tr>
                <th>Hora inicio</th>
                <td>${document.getElementById('inicio').value}</td>
            </tr>

            <tr>
                <th>Hora término</th>
                <td>${document.getElementById('termino').value}</td>
            </tr>

            <tr>
                <th>Tipo de evento</th>
                <td>${selectedEvents.join(', ')}</td>
            </tr>

        </table>

        <h4 style="margin-top:25px;">Participantes</h4>

        <table class="pdf-table">

            <thead>
                <tr>
                    <th>Categoría</th>
                    <th>Mujeres</th>
                    <th>Hombres</th>
                    <th>Total</th>
                </tr>
            </thead>

            <tbody>

                ${participantRows}

            </tbody>

        </table>

        <h4 style="margin-top:25px;">Comentarios del Evento</h4>

        <div style="border:1px solid black;padding:15px;min-height:120px;">
            ${document.getElementById('comentarios').value}
        </div>

        <div class="pdf-signatures">

            <div class="pdf-signature">
                <hr>
                Responsable<br>
                Nombre y firma
            </div>

            <div class="pdf-signature">
                <hr>
                Organizador<br>
                Nombre y firma
            </div>

            <div class="pdf-signature">
                <hr>
                Vo.Bo.<br>
                Nombre y firma
            </div>

        </div>

    </div>

    `;

    const tempDiv = document.createElement('div');

    tempDiv.innerHTML = pdfContent;

    document.body.appendChild(tempDiv);

    html2pdf()
    .set({
        margin:0.3,
        filename:'registro_cultura_fisica.pdf',
        image:{ type:'jpeg', quality:1 },
        html2canvas:{ scale:3 },
        jsPDF:{
            unit:'in',
            format:'letter',
            orientation:'portrait'
        }
    })
    .from(tempDiv)
    .save()
    .then(()=>{
        document.body.removeChild(tempDiv);
    });

}

</script>
